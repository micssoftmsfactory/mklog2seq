var mov_array = new Array();
function make_movno()
{
    var px = 0;
    var str = "";
    for (var i = 0; i < 100; i++) {
        if (mov_array[i]) {
            str += ".movno_r_" + i + " {left:" + px + "}"
            str += ".movno_l_" + i + " {left:-" + px + "}"
            str += ".call_width_" + i + " {width:" + px + "}";
        };
        px += 150;
    }
    str += "";
    $("style").append(str);
}
